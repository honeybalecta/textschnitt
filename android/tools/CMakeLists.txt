cmake_minimum_required(VERSION 3.22)
project(wtest CXX)
set(WHISPER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/whisper.cpp)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(${WHISPER_DIR} whisper EXCLUDE_FROM_ALL)
add_executable(wtest wtest.cpp ../app/src/main/cpp/transcribe.cpp)
target_include_directories(wtest PRIVATE ../app/src/main/cpp)
target_link_libraries(wtest PRIVATE whisper)
